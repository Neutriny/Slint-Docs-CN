import { GroupBox , VerticalBox, CheckBox } from "std-widgets.slint";
export component Example inherits Window {
    width: 200px;
    height: 100px;
    GroupBox {
        title: "Groceries";
        VerticalLayout {
            CheckBox { text: "Bread"; checked: true ;}
            CheckBox { text: "Fruits"; }
        }
    }
}

`GroupBox` 是一个容器，将其子元素归组在一个公共标题下。

## 属性

### content-padding

<SlintProperty
  propName="content-padding"
  typeName="length"
  defaultValue="Depends on the style"
>
  内容整体布局的内边距。此单一值 应用于所有边。
</SlintProperty>

### enabled

<SlintProperty propName="enabled" typeName="bool" defaultValue="true">
  当为 false 时，无法与分组框交互
</SlintProperty>

### title

<SlintProperty propName="title" typeName="string">
  作为分组框标题显示的文本。
</SlintProperty>
